		formatString: #,0
		displayFolder: 1 Passenger KPIs

	measure 'High Tier Rate' = DIVIDE([High Tier Passengers], [Total Passengers], 0)
		formatString: 0.0%
		displayFolder: 1 Passenger KPIs

	measure 'Low Tier Rate' = DIVIDE([Low Tier Passengers], [Total Passengers], 0)
		formatString: 0.0%

	measure 'Share of Passengers' = DIVIDE([Total Passengers], CALCULATE([Total Passengers], ALLSELECTED()), 0)
		formatString: 0.0%
		displayFolder: 1 Passenger KPIs

	/// Share of a category's passengers in each satisfaction bucket - bars always add to 100%.

		displayFolder: 1 Passenger KPIs

		formatString: 0.0%
		displayFolder: 1 Passenger KPIs

	measure 'Satisfaction Gap vs Overall (pts)' = ([Satisfaction Rate] - [Overall Satisfaction Rate]) * 100
		formatString: +0.0;-0.0;0.0
		displayFolder: 1 Passenger KPIs

	measure 'Avg Departure Delay' = AVERAGE(FactPassenger[Departure Delay in Minutes])
		formatString: 0.0
		displayFolder: 2 Delay & Distance

		formatString: #,0
		displayFolder: 2 Delay & Distance

	measure 'Avg Passenger Age' = AVERAGE(FactPassenger[Age])
		formatString: 0.0
		displayFolder: 2 Delay & Distance

	measure 'Delayed Share' = DIVIDE(CALCULATE([Total Passengers], FactPassenger[Arrival Delay in Minutes] > 15), [Total Passengers], 0)
		formatString: 0.0%
		displayFolder: 2 Delay & Distance

		formatString: 0.0%
		displayFolder: 2 Delay & Distance

	measure 'Satisfaction Rate - Delayed' = CALCULATE([Satisfaction Rate], FactPassenger[Arrival Delay in Minutes] > 15)
		formatString: 0.0%
		displayFolder: 2 Delay & Distance

	measure 'Delay Satisfaction Penalty (pts)' = ([Satisfaction Rate - Delayed] - [Satisfaction Rate - On Time]) * 100
		formatString: +0.0;-0.0;0.0
		displayFolder: 2 Delay & Distance

	measure 'Rating Gap (Satisfied - Not)' = [Avg Rating - Satisfied] - [Avg Rating - Neutral or Dissatisfied]

	measure 'Lowest Rated Attribute' =

			VAR t = ADDCOLUMNS(VALUES(DimServiceAttribute[Service Attribute]), "@v", [Average Service Rating])
